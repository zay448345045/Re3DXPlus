// attribs: aPos aUV aColor
//----VS
attribute vec3 aPos; attribute vec2 aUV; attribute vec4 aColor;
varying vec2 vUV; varying lowp vec4 vColor;
void main() { gl_Position = vec4(aPos, 1.0); vUV = aUV; vColor = aColor; }

//----FS
precision mediump float;
uniform sampler2D uTex;
varying vec2 vUV; varying lowp vec4 vColor;
void main() { gl_FragColor = texture2D(uTex, vUV) * vColor; }
