// attribs: aPos
//----VS
attribute vec2 aPos; varying vec2 vUV;
void main() { gl_Position = vec4(aPos, 0.0, 1.0); vUV = aPos * 0.5 + 0.5; }

//----FS
#define BRIGHT 1
precision mediump float; uniform sampler2D uTex; uniform vec2 uTexel; varying vec2 vUV;
#if BRIGHT
uniform float uThresh;
#endif
void main() {
  vec3 c = (texture2D(uTex, vUV + vec2(-uTexel.x, -uTexel.y)).rgb + texture2D(uTex, vUV + vec2(uTexel.x, -uTexel.y)).rgb
          + texture2D(uTex, vUV + vec2(-uTexel.x, uTexel.y)).rgb + texture2D(uTex, vUV + uTexel).rgb) * 0.25;
#if BRIGHT
  float br = max(c.r, max(c.g, c.b)), knee = uThresh * 0.5;
  float soft = clamp(br - uThresh + knee, 0.0, 2.0 * knee);
  soft = soft * soft / (4.0 * knee + 0.0001);
  c *= max(soft, br - uThresh) / max(br, 0.0001);
#endif
  gl_FragColor = vec4(c, 1.0);
}
