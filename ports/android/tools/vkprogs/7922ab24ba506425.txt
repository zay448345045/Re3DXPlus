// attribs: aPos aUV
//----VS
attribute vec2 aPos; attribute vec2 aUV; varying vec2 vUV;
uniform vec2 uView;
void main() { vUV = aUV; gl_Position = vec4(aPos.x * uView.x + uView.y, 1.0 - aPos.y / 384.0, 0.0, 1.0); }

//----FS
precision mediump float; uniform sampler2D uTex; uniform vec4 uMul, uAdd; varying vec2 vUV;
void main() { gl_FragColor = clamp(texture2D(uTex, vUV) * uMul + uAdd, 0.0, 1.0); }
