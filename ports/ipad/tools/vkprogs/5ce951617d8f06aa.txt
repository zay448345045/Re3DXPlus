// attribs: aPos
//----VS
attribute vec2 aPos; varying vec2 vUV;
void main() { gl_Position = vec4(aPos, 0.0, 1.0); vUV = aPos * 0.5 + 0.5; }

//----FS
precision mediump float; uniform sampler2D uTex; uniform vec2 uTexel; varying vec2 vUV;
void main() {
  vec2 d = uTexel;
  vec3 c = texture2D(uTex, vUV).rgb * 4.0
         + (texture2D(uTex, vUV + vec2(d.x, 0.0)).rgb + texture2D(uTex, vUV - vec2(d.x, 0.0)).rgb
         + texture2D(uTex, vUV + vec2(0.0, d.y)).rgb + texture2D(uTex, vUV - vec2(0.0, d.y)).rgb) * 2.0
         + texture2D(uTex, vUV + d).rgb + texture2D(uTex, vUV - d).rgb
         + texture2D(uTex, vUV + vec2(d.x, -d.y)).rgb + texture2D(uTex, vUV + vec2(-d.x, d.y)).rgb;
  gl_FragColor = vec4(c * (1.0 / 16.0), 1.0);
}
